function(watt_add_test name)
    qt_add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE Qt6::Test ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

watt_add_test(tst_workout        watt_core)
watt_add_test(tst_engine         watt_core)
watt_add_test(tst_analysis       watt_core)
watt_add_test(tst_database       watt_core)
watt_add_test(tst_session        watt_core)
watt_add_test(tst_builder        watt_core)
watt_add_test(tst_bleprotocol    watt_bleprotocol)
